40/* Binary min-heap of (degree, node) with lazy deletion: stale items are skipped when popped. */
41typedef struct {
42 int degree;
43 int node;
44} HeapItem;

46typedef struct {
47 HeapItem* items;
48 size_t count;
49 size_t capacity;
50} Heap;

52static bool heap_less(HeapItem a, HeapItem b) { return a.degree < b.degree || (a.degree == b.degree && a.node < b.node); }
53
54static bool heap_push(Heap* h, int degree, int node) {
55 if (h->count == h->capacity) {
56 const size_t capacity = h->capacity ? h->capacity * 2 : 64;
57 HeapItem* grown = (HeapItem*)realloc(h->items, capacity * sizeof(HeapItem));
58 if (grown == NULL) {
59 return false;
60 }
61 h->items = grown;
62 h->capacity = capacity;
63 }
64 size_t i = h->count++;
65 const HeapItem item = {degree, node};
66 while (i > 0 && heap_less(item, h->items[(i - 1) / 2])) {
67 h->items[i] = h->items[(i - 1) / 2];
68 i = (i - 1) / 2;
69 }
70 h->items[i] = item;
71 return true;
72}
73
74static HeapItem heap_pop(Heap* h) {
75 const HeapItem top = h->items[0];
76 const HeapItem last = h->items[--h->count];
77 size_t i = 0;
78 for (;;) {
79 size_t child = 2 * i + 1;
80 if (child >= h->count) {
81 break;
82 }
83 if (child + 1 < h->count && heap_less(h->items[child + 1], h->items[child])) {
84 child++;
85 }
86 if (!heap_less(h->items[child], last)) {
87 break;
88 }
89 h->items[i] = h->items[child];
90 i = child;
91 }
92 if (h->count > 0) {
93 h->items[i] = last;
94 }
95 return top;
96}

122/*
123 * Minimum-degree ordering on the explicit elimination graph: eliminate the
124 * node with the fewest neighbours, connect its neighbours to each other (the
125 * fill) and record them as the pattern of its column of L. Networks of a
126 * house are close to trees, so the fill stays small. The node with the
127 * smallest degree comes from a heap (ties: lowest index, so the ordering is
128 * reproducible); done once per topology.
129 */
130ElSparse* el_sparse_analyze(int n, const int* pair_a, const int* pair_b, size_t pairs) {
131 ElSparse* s = (ElSparse*)calloc(1, sizeof(ElSparse));
132 if (s == NULL) {
133 return NULL;
134 }
135 s->n = n;
136 s->pairs = pairs;
137 const size_t un = (size_t)(n > 0 ? n : 1);
138 IntList* adjacency = (IntList*)calloc(un, sizeof(IntList));
139 IntList* columns = (IntList*)calloc(un, sizeof(IntList)); /* original indices */
140 unsigned char* eliminated = (unsigned char*)calloc(un, 1);
141 int* mark = (int*)malloc(un * sizeof(int));
142 s->perm = (int*)malloc(un * sizeof(int));
143 s->iperm = (int*)malloc(un * sizeof(int));
144 bool ok = adjacency && columns && eliminated && mark && s->perm && s->iperm;
145
146 for (size_t p = 0; ok && p < pairs; p++) {
147 ok = list_push(&adjacency[pair_a[p]], pair_b[p]) && list_push(&adjacency[pair_b[p]], pair_a[p]);
148 }
149 Heap heap = {NULL, 0, 0};
150 for (int i = 0; ok && i < n; i++) {
151 mark[i] = -1;
152 ok = heap_push(&heap, adjacency[i].count, i);
153 }
154 for (int k = 0; ok && k < n; k++) {
155 int v = -1;
156 while (v < 0 && heap.count > 0) {
157 const HeapItem item = heap_pop(&heap);
158 if (!eliminated[item.node] && item.degree == adjacency[item.node].count) {
159 v = item.node;
160 }
161 }
162 if (v < 0) {
163 ok = false;
164 break;
165 }
166 eliminated[v] = 1;
167 s->perm[k] = v;
168 s->iperm[v] = k;
169 const IntList* neighbours = &adjacency[v];
170 for (int a = 0; ok && a < neighbours->count; a++) {
171 const int u = neighbours->items[a];
172 ok = list_push(&columns[v], u);
173 list_remove(&adjacency[u], v);
174 /* u gets every other neighbour of v (fill) */
175 for (int b = 0; b < adjacency[u].count; b++) {
176 mark[adjacency[u].items[b]] = u;
177 }
178 for (int b = 0; ok && b < neighbours->count; b++) {
179 const int w = neighbours->items[b];
180 if (w != u && mark[w] != u) {
181 mark[w] = u;
182 ok = list_push(&adjacency[u], w);
183 }
184 }
185 ok = ok && heap_push(&heap, adjacency[u].count, u); /* its degree changed */
186 }
187 free(adjacency[v].items);
188 memset(&adjacency[v], 0, sizeof(IntList));
189 }
190 free(heap.items);
191
192 /* Column pattern in permuted numbering, rows ascending. */
193 size_t nonzeros = 0;
194 for (int i = 0; ok && i < n; i++) {
195 nonzeros += (size_t)columns[i].count;
196 }
197 const size_t unz = nonzeros ? nonzeros : 1;
198 if (ok) {
199 s->col_start = (int*)malloc((un + 1) * sizeof(int));
200 s->row_index = (int*)malloc(unz * sizeof(int));
201 s->values = (ElPhasor*)calloc(unz, sizeof(ElPhasor));
202 s->diag = (ElPhasor*)calloc(un, sizeof(ElPhasor));
203 s->work = (ElPhasor*)calloc(un, sizeof(ElPhasor));
204 s->row_start = (int*)calloc(un + 1, sizeof(int));
205 s->row_entry = (int*)malloc(unz * sizeof(int));
206 s->row_col = (int*)malloc(unz * sizeof(int));
207 s->pair_pos = (int*)malloc((pairs ? pairs : 1) * sizeof(int));
208 ok = s->col_start && s->row_index && s->values && s->diag && s->work && s->row_start && s->row_entry &&
209 s->row_col && s->pair_pos;
210 }
211 if (ok) {
212 int position = 0;
213 for (int k = 0; k < n; k++) {
214 const IntList* column = &columns[s->perm[k]];
215 s->col_start[k] = position;
216 for (int e = 0; e < column->count; e++) {
217 s->row_index[position + e] = s->iperm[column->items[e]];
218 }
219 qsort(s->row_index + position, (size_t)column->count, sizeof(int), compare_int);
220 position += column->count;
221 }
222 s->col_start[n] = position;
223
224 /* Row lists: for row j the entries L(j,k), k < j (columns in ascending order). */
225 for (int p = 0; p < position; p++) {
226 s->row_start[s->row_index[p] + 1]++;
227 }
228 for (int j = 0; j < n; j++) {
229 s->row_start[j + 1] += s->row_start[j];
230 }
231 for (int j = 0; j < n; j++) {
232 mark[j] = s->row_start[j];
233 }
234 for (int k = 0; k < n; k++) {
235 for (int p = s->col_start[k]; p < s->col_start[k + 1]; p++) {
236 const int j = s->row_index[p];
237 s->row_entry[mark[j]] = p;
238 s->row_col[mark[j]] = k;
239 mark[j]++;
240 }
241 }
242
243 /* Where every input pair lands in L. */
244 for (size_t p = 0; p < pairs; p++) {
245 const int ia = s->iperm[pair_a[p]];
246 const int ib = s->iperm[pair_b[p]];
247 const int column = ia < ib ? ia : ib;
248 const int row = ia < ib ? ib : ia;
249 const int* first = s->row_index + s->col_start[column];
250 const int* found = (const int*)bsearch(&row, first, (size_t)(s->col_start[column + 1] - s->col_start[column]),
251 sizeof(int), compare_int);
252 s->pair_pos[p] = found ? (int)(found - s->row_index) : -1; /* always found: A is part of L's pattern */
253 }
254 }
255
256 for (int i = 0; adjacency && i < n; i++) {
257 free(adjacency[i].items);
258 }
259 for (int i = 0; columns && i < n; i++) {
260 free(columns[i].items);
261 }
262 free(adjacency);
263 free(columns);
264 free(eliminated);
265 free(mark);
266 if (!ok) {
267 el_sparse_free(s);
268 return NULL;
269 }
270 return s;
271}

293/*
294 * Left-looking LDL^T in place: column j is gathered into the dense work
295 * vector, updated with every earlier column k that has an entry in row j
296 * (w -= L(:,k) L(j,k) D(k)), then divided by its pivot D(j). The filled
297 * pattern guarantees that every updated row is part of column j.
298 */
299bool el_sparse_factor(ElSparse* s, double min_pivot) {
300 ElPhasor* w = s->work;
301 s->perturbed = 0;
302 for (int j = 0; j < s->n; j++) {
303 w[j] = s->diag[j];
304 for (int p = s->col_start[j]; p < s->col_start[j + 1]; p++) {
305 w[s->row_index[p]] = s->values[p];
306 }
307 for (int q = s->row_start[j]; q < s->row_start[j + 1]; q++) {
308 const int k = s->row_col[q];
309 const int entry = s->row_entry[q]; /* L(j,k); the rows after it in column k are > j */
310 const ElPhasor factor = el_mul(s->values[entry], s->diag[k]);
311 for (int p = entry; p < s->col_start[k + 1]; p++) {
312 ElPhasor* target = &w[s->row_index[p]];
313 *target = el_sub(*target, el_mul(s->values[p], factor));
314 }
315 }
316 ElPhasor pivot = w[j];
317 if (!isfinite(pivot.re) || !isfinite(pivot.im)) {
318 return false;
319 }
320 if (!(pivot.re > 0.0)) {
321 pivot = el_phasor(min_pivot, 0.0);
322 s->perturbed++;
323 }
324 s->diag[j] = pivot;
325 w[j] = el_phasor(0.0, 0.0);
326 for (int p = s->col_start[j]; p < s->col_start[j + 1]; p++) {
327 const int i = s->row_index[p];
328 s->values[p] = el_div(w[i], pivot);
329 w[i] = el_phasor(0.0, 0.0);
330 }
331 }
332 return true;
333}

335void el_sparse_solve(ElSparse* s, ElPhasor* x) {
336 ElPhasor* y = s->work;
337 const int n = s->n;
338 for (int k = 0; k < n; k++) {
339 y[k] = x[s->perm[k]];
340 }
341 for (int k = 0; k < n; k++) { /* L z = b */
342 const ElPhasor yk = y[k];
343 if (yk.re == 0.0 && yk.im == 0.0) {
344 continue;
345 }
346 for (int p = s->col_start[k]; p < s->col_start[k + 1]; p++) {
347 ElPhasor* target = &y[s->row_index[p]];
348 *target = el_sub(*target, el_mul(s->values[p], yk));
349 }
350 }
351 for (int k = 0; k < n; k++) { /* D */
352 y[k] = el_div(y[k], s->diag[k]);
353 }
354 for (int k = n - 1; k >= 0; k--) { /* L^T x = y */
355 ElPhasor sum = y[k];
356 for (int p = s->col_start[k]; p < s->col_start[k + 1]; p++) {
357 sum = el_sub(sum, el_mul(s->values[p], y[s->row_index[p]]));
358 }
359 y[k] = sum;
360 }
361 for (int k = 0; k < n; k++) {
362 x[s->perm[k]] = y[k];
363 y[k] = el_phasor(0.0, 0.0); /* the work vector is zero between uses */
364 }
