96bool el_circuit_solve(ElCircuit* circuit, double gmin) {
97 circuit->solved = false;
98 const int nodes = circuit->nodes - 1; /* unknown node voltages (without earth) */
99 int sources = 0;
100 for (size_t i = 0; i < circuit->count; i++) {
101 sources += circuit->elements[i].kind == EL_VOLTAGE_SOURCE;
102 }
103 const int n = nodes + sources;
104
105 /* Node voltages: allocated once per node count (nodes never change between solutions). */
106 if (circuit->voltages == NULL) {
107 circuit->voltages = (ElPhasor*)calloc((size_t)circuit->nodes, sizeof(ElPhasor));
108 }
109 const size_t elements = circuit->count ? circuit->count : 1;
110 if (circuit->voltages == NULL || !reserve(&circuit->source_currents, &circuit->result_capacity, elements)) {
111 return false;
112 }
113 memset(circuit->voltages, 0, (size_t)circuit->nodes * sizeof(ElPhasor));
114 memset(circuit->source_currents, 0, elements * sizeof(ElPhasor));
115 if (n == 0) {
116 circuit->solved = true;
117 return true;
118 }
119
120 /* Augmented matrix n x (n + 1), row-major. */
121 const size_t width = (size_t)n + 1;
122 if (!reserve(&circuit->matrix, &circuit->matrix_capacity, (size_t)n * width)) {
123 return false;
124 }
125 ElPhasor* m = circuit->matrix;
126 memset(m, 0, (size_t)n * width * sizeof(ElPhasor));
127#define M(r, c) m[(size_t)(r) * width + (size_t)(c)]
128 for (int k = 0; k < nodes; k++) {
129 M(k, k).re += gmin;
130 }
131 int source_row = nodes;
132 for (size_t i = 0; i < circuit->count; i++) {
133 const ElElement* e = &circuit->elements[i];
134 const int a = e->a - 1; /* -1 = earth */
135 const int b = e->b - 1;
136 switch (e->kind) {
137 case EL_ADMITTANCE:
138 if (a >= 0) {
139 M(a, a) = el_add(M(a, a), e->value);
140 }
141 if (b >= 0) {
142 M(b, b) = el_add(M(b, b), e->value);
143 }
144 if (a >= 0 && b >= 0) {
145 M(a, b) = el_sub(M(a, b), e->value);
146 M(b, a) = el_sub(M(b, a), e->value);
147 }
148 break;
149 case EL_CURRENT_SOURCE:
150 /* value leaves node a and enters node b */
151 if (a >= 0) {
152 M(a, n) = el_sub(M(a, n), e->value);
153 }
154 if (b >= 0) {
155 M(b, n) = el_add(M(b, n), e->value);
156 }
157 break;
158 case EL_VOLTAGE_SOURCE: {
159 const int r = source_row++;
160 /* KCL: the source current j (a -> b) leaves node a and enters node b */
161 if (a >= 0) {
162 M(a, r).re += 1.0;
163 M(r, a).re += 1.0;
164 }
165 if (b >= 0) {
166 M(b, r).re -= 1.0;
167 M(r, b).re -= 1.0;
168 }
169 M(r, n) = e->value;
170 break;
171 }
172 }
173 }
174
175 /* Gaussian elimination with partial pivoting. */
176 bool ok = true;
177 for (int col = 0; col < n && ok; col++) {
178 int pivot = col;
179 double best = el_abs2(M(col, col));
180 for (int r = col + 1; r < n; r++) {
181 const double magnitude = el_abs2(M(r, col));
182 if (magnitude > best) {
183 best = magnitude;
184 pivot = r;
185 }
186 }
187 if (best < 1e-36) {
188 ok = false;
189 break;
190 }
191 if (pivot != col) {
192 for (int c = col; c <= n; c++) {
193 const ElPhasor t = M(col, c);
194 M(col, c) = M(pivot, c);
195 M(pivot, c) = t;
196 }
197 }
198 const ElPhasor diagonal = M(col, col);
199 for (int r = col + 1; r < n; r++) {
200 const ElPhasor below = M(r, col);
201 if (below.re == 0.0 && below.im == 0.0) {
202 continue;
203 }
204 const ElPhasor factor = el_div(below, diagonal);
205 for (int c = col; c <= n; c++) {
206 M(r, c) = el_sub(M(r, c), el_mul(factor, M(col, c)));
207 }
208 }
209 }
210 if (ok) {
211 for (int r = n - 1; r >= 0; r--) {
212 ElPhasor sum = M(r, n);
213 for (int c = r + 1; c < n; c++) {
214 sum = el_sub(sum, el_mul(M(r, c), M(c, n))); /* M(c, n) holds the solution of row c */
215 }
216 M(r, n) = el_div(sum, M(r, r));
217 }
218 for (int k = 0; k < nodes; k++) {
219 circuit->voltages[k + 1] = M(k, n);
220 }
221 source_row = nodes;
222 for (size_t i = 0; i < circuit->count; i++) {
223 if (circuit->elements[i].kind == EL_VOLTAGE_SOURCE) {
224 circuit->source_currents[i] = M(source_row, n);
225 source_row++;
226 }
227 }
228 circuit->solved = true;
229 }
230#undef M
231 return ok;
232}
