15/*
16 * A few topologies are in use at the same time (the network itself, a dimmer
17 * measuring its surroundings, a measurement), so several analyses are kept.
18 * An entry is identified by its admittance elements (nodes in order); it
19 * keeps the values of its last factorization to skip refactorizing.

68static int compare_pair(const void* a, const void* b) {
69 const PairKey* x = (const PairKey*)a;
70 const PairKey* y = (const PairKey*)b;

72 return (x->lo > y->lo) - (x->lo < y->lo);

74 return (x->hi > y->hi) - (x->hi < y->hi);

96static bool entry_build(SparseEntry* e, const ElCircuit* circuit, size_t admittances) {
97 entry_free(e);
98 const int n = circuit->nodes - 1;
99 const size_t count = admittances ? admittances : 1;
100 e->nodes = (int*)malloc(2 * count * sizeof(int));
101 e->pair = (int*)malloc(count * sizeof(int));
102 e->values = (ElPhasor*)malloc(count * sizeof(ElPhasor));
103 PairKey* keys = (PairKey*)malloc(count * sizeof(PairKey));
104 int* pair_a = (int*)malloc(count * sizeof(int));
105 int* pair_b = (int*)malloc(count * sizeof(int));
106 bool ok = e->nodes && e->pair && e->values && keys && pair_a && pair_b;
107 size_t key_count = 0;
108 size_t j = 0;
109 for (size_t i = 0; ok && i < circuit->count; i++) {
110 const ElElement* element = &circuit->elements[i];
111 if (element->kind != EL_ADMITTANCE) {
112 continue;
113 }
114 e->nodes[2 * j] = element->a;
115 e->nodes[2 * j + 1] = element->b;
116 e->pair[j] = -1;
117 if (element->a > 0 && element->b > 0 && element->a != element->b) {
118 const int a = element->a - 1;
119 const int b = element->b - 1;
120 keys[key_count].lo = a < b ? a : b;
121 keys[key_count].hi = a < b ? b : a;
122 keys[key_count].element = (int)j;
123 key_count++;
124 }
125 j++;
126 }
127 size_t pairs = 0;
128 if (ok) {
129 qsort(keys, key_count, sizeof(PairKey), compare_pair);
130 for (size_t k = 0; k < key_count; k++) {
131 if (pairs == 0 || keys[k].lo != pair_a[pairs - 1] || keys[k].hi != pair_b[pairs - 1]) {
132 pair_a[pairs] = keys[k].lo;
133 pair_b[pairs] = keys[k].hi;
134 pairs++;
135 }
136 e->pair[keys[k].element] = (int)pairs - 1;
137 }
138 e->sparse = el_sparse_analyze(n, pair_a, pair_b, pairs);
139 ok = e->sparse != NULL;
140 }
141 free(keys);
142 free(pair_a);
143 free(pair_b);
144 if (!ok) {
145 entry_free(e);
146 return false;
147 }
148 e->count = admittances;
149 return true;

170static bool entry_factor(SparseEntry* e, ElCircuit* circuit, double gmin) {
171 ElSparse* s = e->sparse;
172 el_sparse_clear(s);
173 for (int k = 0; k < s->n; k++) {
174 el_sparse_add_diagonal(s, k, el_phasor(gmin, 0.0));
175 }
176 size_t j = 0;
177 for (size_t i = 0; i < circuit->count; i++) {
178 const ElElement* element = &circuit->elements[i];
179 if (element->kind != EL_ADMITTANCE) {
180 continue;
181 }
182 const ElPhasor y = element->value;
183 e->values[j] = y;
184 j++;
185 if (element->a == element->b) {
186 continue; /* from a node to itself: no effect */
187 }
188 if (element->a > 0) {
189 el_sparse_add_diagonal(s, element->a - 1, y);
190 }
191 if (element->b > 0) {
192 el_sparse_add_diagonal(s, element->b - 1, y);
193 }
194 if (e->pair[j - 1] >= 0) {
195 el_sparse_add_pair(s, (size_t)e->pair[j - 1], el_scale(y, -1.0));
196 }
197 }
198 e->gmin = gmin;
199 e->factored = el_sparse_factor(s, gmin > 0.0 ? gmin : 1e-12);

206static bool solve_sparse(ElCircuit* circuit, double gmin) {
207 const int n = circuit->nodes - 1;
208 SparseCache* cache = (SparseCache*)circuit->sparse_cache;
209 if (cache == NULL) {
210 cache = (SparseCache*)calloc(1, sizeof(SparseCache));
211 if (cache == NULL) {
212 return false;
213 }
214 circuit->sparse_cache = cache;
215 }
216 size_t admittances = 0;
217 for (size_t i = 0; i < circuit->count; i++) {
218 admittances += circuit->elements[i].kind == EL_ADMITTANCE;
219 }
220
221 SparseEntry* e = NULL;
222 SparseEntry* oldest = &cache->entries[0];
223 for (int i = 0; i < SPARSE_CACHE_ENTRIES; i++) {
224 SparseEntry* candidate = &cache->entries[i];
225 if (same_topology(candidate, circuit, admittances)) {
226 e = candidate;
227 break;
228 }
229 if (candidate->used < oldest->used) {
230 oldest = candidate;
231 }
232 }
233 if (e == NULL) {
234 e = oldest;
235 if (!entry_build(e, circuit, admittances)) {
236 return false;
237 }
238 circuit->analyses++;
239 }
240 e->used = ++cache->clock;
241 if (!values_unchanged(e, circuit, gmin) && !entry_factor(e, circuit, gmin)) {
242 return false;
243 }
244
245 const size_t size = (size_t)(n > 0 ? n : 1);
246 if (size > cache->rhs_capacity || cache->rhs == NULL) {
247 ElPhasor* grown = (ElPhasor*)realloc(cache->rhs, size * sizeof(ElPhasor));
248 if (grown == NULL) {
249 return false;
250 }
251 cache->rhs = grown;
252 cache->rhs_capacity = size;
253 }
254 ElPhasor* x = cache->rhs;
255 memset(x, 0, size * sizeof(ElPhasor));
256 for (size_t i = 0; i < circuit->count; i++) {
257 const ElElement* element = &circuit->elements[i];
258 if (element->kind == EL_CURRENT_SOURCE) {
259 /* value leaves node a and enters node b */
260 if (element->a > 0) {
261 x[element->a - 1] = el_sub(x[element->a - 1], element->value);
262 }
263 if (element->b > 0) {
264 x[element->b - 1] = el_add(x[element->b - 1], element->value);
265 }
266 }
267 }
268 el_sparse_solve(e->sparse, x);
269 for (int k = 0; k < n; k++) {
270 circuit->voltages[k + 1] = x[k];
271 }
272 circuit->fill = el_sparse_nonzeros(e->sparse);
273 circuit->sparse_solutions++;
274 return true;

365bool el_circuit_solve(ElCircuit* circuit, double gmin) {
366 circuit->solved = false;
367 const int nodes = circuit->nodes - 1; /* unknown node voltages (without earth) */
368 int sources = 0;
369 for (size_t i = 0; i < circuit->count; i++) {
370 sources += circuit->elements[i].kind == EL_VOLTAGE_SOURCE;
371 }
372 const int n = nodes + sources;
373
374 /* Node voltages: allocated once per node count (nodes never change between solutions). */
375 if (circuit->voltages == NULL) {
376 circuit->voltages = (ElPhasor*)calloc((size_t)circuit->nodes, sizeof(ElPhasor));
377 }
378 const size_t elements = circuit->count ? circuit->count : 1;
379 if (circuit->voltages == NULL || !reserve(&circuit->source_currents, &circuit->result_capacity, elements)) {
380 return false;
381 }
382 memset(circuit->voltages, 0, (size_t)circuit->nodes * sizeof(ElPhasor));
383 memset(circuit->source_currents, 0, elements * sizeof(ElPhasor));
384 if (n == 0) {
385 circuit->solved = true;
386 return true;
387 }
388 if (sources == 0 && circuit->solver == EL_SOLVER_AUTO && solve_sparse(circuit, gmin)) {
389 circuit->solved = true;
390 return true;
391 }
392 circuit->dense_solutions++;
393
394 /* Dense modified nodal analysis. Augmented matrix n x (n + 1), row-major. */
395 const size_t width = (size_t)n + 1;
396 if (!reserve(&circuit->matrix, &circuit->matrix_capacity, (size_t)n * width)) {
397 return false;
398 }
399 ElPhasor* m = circuit->matrix;
400 memset(m, 0, (size_t)n * width * sizeof(ElPhasor));
401#define M(r, c) m[(size_t)(r) * width + (size_t)(c)]
402 for (int k = 0; k < nodes; k++) {
403 M(k, k).re += gmin;
404 }
405 int source_row = nodes;
406 for (size_t i = 0; i < circuit->count; i++) {
407 const ElElement* e = &circuit->elements[i];
408 const int a = e->a - 1; /* -1 = earth */
409 const int b = e->b - 1;
410 switch (e->kind) {
411 case EL_ADMITTANCE:
412 if (a >= 0) {
413 M(a, a) = el_add(M(a, a), e->value);
414 }
415 if (b >= 0) {
416 M(b, b) = el_add(M(b, b), e->value);
417 }
418 if (a >= 0 && b >= 0) {
419 M(a, b) = el_sub(M(a, b), e->value);
420 M(b, a) = el_sub(M(b, a), e->value);
421 }
422 break;
423 case EL_CURRENT_SOURCE:
424 /* value leaves node a and enters node b */
425 if (a >= 0) {
426 M(a, n) = el_sub(M(a, n), e->value);
427 }
428 if (b >= 0) {
429 M(b, n) = el_add(M(b, n), e->value);
430 }
431 break;
432 case EL_VOLTAGE_SOURCE: {
433 const int r = source_row++;
434 /* KCL: the source current j (a -> b) leaves node a and enters node b */
435 if (a >= 0) {
436 M(a, r).re += 1.0;
437 M(r, a).re += 1.0;
438 }
439 if (b >= 0) {
440 M(b, r).re -= 1.0;
441 M(r, b).re -= 1.0;
442 }
443 M(r, n) = e->value;
444 break;
445 }
446 }
447 }
448
449 /* Gaussian elimination with partial pivoting. */
450 bool ok = true;
451 for (int col = 0; col < n && ok; col++) {
452 int pivot = col;
453 double best = el_abs2(M(col, col));
454 for (int r = col + 1; r < n; r++) {
455 const double magnitude = el_abs2(M(r, col));
456 if (magnitude > best) {
457 best = magnitude;
458 pivot = r;
459 }
460 }
461 if (best < 1e-36) {
462 ok = false;
463 break;
464 }
465 if (pivot != col) {
466 for (int c = col; c <= n; c++) {
467 const ElPhasor t = M(col, c);
468 M(col, c) = M(pivot, c);
469 M(pivot, c) = t;
470 }
471 }
472 const ElPhasor diagonal = M(col, col);
473 for (int r = col + 1; r < n; r++) {
474 const ElPhasor below = M(r, col);
475 if (below.re == 0.0 && below.im == 0.0) {
476 continue;
477 }
478 const ElPhasor factor = el_div(below, diagonal);
479 for (int c = col; c <= n; c++) {
480 M(r, c) = el_sub(M(r, c), el_mul(factor, M(col, c)));
481 }
482 }
483 }
484 if (ok) {
485 for (int r = n - 1; r >= 0; r--) {
486 ElPhasor sum = M(r, n);
487 for (int c = r + 1; c < n; c++) {
488 sum = el_sub(sum, el_mul(M(r, c), M(c, n))); /* M(c, n) holds the solution of row c */
489 }
490 M(r, n) = el_div(sum, M(r, r));
491 }
492 for (int k = 0; k < nodes; k++) {
493 circuit->voltages[k + 1] = M(k, n);
494 }
495 source_row = nodes;
496 for (size_t i = 0; i < circuit->count; i++) {
497 if (circuit->elements[i].kind == EL_VOLTAGE_SOURCE) {
498 circuit->source_currents[i] = M(source_row, n);
499 source_row++;
500 }
501 }
502 circuit->solved = true;
503 }
504#undef M
505 return ok;
506}
