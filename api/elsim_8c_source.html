179int el_model_add_device(ElModel* model, const char* id, ElDeviceType type) {
180 if (type < 0 || type >= EL_DEVICE_TYPES || el_model_find_device(model, id) >= 0) {
181 return -1;
182 }
183 ElDevice* d = grow((void**)&model->devices, &model->device_count, sizeof(ElDevice));

185 return -1;
186 }
187 copy_id(d->id, sizeof(d->id), id);
188 d->type = type;
189 d->first_node = model->node_count;
190 model->node_count += kTerminalCount[type];
191 d->internal_node = -1;

193 d->internal_node = model->node_count;
194 model->node_count += SUPPLY_INTERNAL_NODES;
195 }
196 set_defaults(d);
197 model->revision++;
198 return (int)model->device_count - 1;

217 const char* colon = reference ? strchr(reference, ':') : NULL;
218 if (colon == NULL) {

220 }

227 device_id[length] = '\0';
228 const int index = el_model_find_device(model, device_id);
229 if (index < 0) {
230 return -1;
231 }
232 const ElDevice* d = &model->devices[index];
233 for (int t = 0; t < kTerminalCount[d->type]; t++) {
234 if (strcmp(kTerminals[d->type][t], colon + 1) == 0) {
235 return d->first_node + t;
236 }
237 }
238 return -1;

432/*
433 * Builds the network of the present state. With zero_sources all voltage
434 * sources are 0 V (for Thevenin measurements); inject_a/inject_b >= 0 add a
435 * 1 A test current that enters node a and leaves node b. The device with
436 * index exclude (>= 0) is left out (a dimmer measuring its surroundings).
437 */
438static bool build_and_solve(ElModel* model, bool zero_sources, int inject_a, int inject_b, int exclude) {
439 ElCircuit* c = &model->circuit;
440 if (c->nodes != model->node_count) {
441 el_circuit_free(c);
442 el_circuit_init(c, model->node_count);
443 }
444 el_circuit_clear(c);
445
446 for (size_t i = 0; i < model->connection_count; i++) {
447 ElConnection* k = &model->connections[i];
448 k->element = k->open ? -1 : el_circuit_add_resistance(c, k->a, k->b, k->resistance_ohm);
449 }
450 for (size_t i = 0; i < model->fault_count; i++) {
451 ElFault* f = &model->faults[i];
452 f->element = el_circuit_add_resistance(c, f->a, f->b, f->resistance_ohm);
453 }
454
455 for (size_t i = 0; i < model->device_count; i++) {
456 ElDevice* d = &model->devices[i];
457 const int t = d->first_node;
458 d->element[0] = d->element[1] = d->element[2] = -1;
459 if ((int)i == exclude) {
460 continue;
461 }
462 switch (d->type) {
463 case EL_SUPPLY: {
464 const int star = d->internal_node;
465 const double half = d->loop_impedance_ohm / 2.0;
466 for (int phase = 0; phase < 3; phase++) {
467 const int source = d->internal_node + 1 + phase;
468 const ElPhasor u = zero_sources ? el_phasor(0.0, 0.0) : el_polar(d->voltage_v, -120.0 * phase);
469 el_circuit_add_voltage_source(c, source, star, u);
470 const int element = el_circuit_add_resistance(c, source, t + SUPPLY_L1 + phase, half);
471 if (phase == 0) {
472 d->element[0] = element;
473 }
474 }
475 d->element[1] = el_circuit_add_resistance(c, t + SUPPLY_N, star, half);
476 el_circuit_add_resistance(c, star, 0, d->source_earth_ohm);
477 if (d->earthing == EL_TT) {
478 d->element[2] = el_circuit_add_resistance(c, t + SUPPLY_PE, 0, d->earth_electrode_ohm);
479 } else {
480 d->element[2] = el_circuit_add_resistance(c, t + SUPPLY_PE, star, half);
481 }
482 break;
483 }
484 case EL_BREAKER:
485 case EL_FUSE:
486 case EL_SWITCH:
487 if (conducting(d)) {
488 d->element[0] = el_circuit_add_resistance(c, t + TWO_IN, t + TWO_OUT, d->contact_ohm);
489 }
490 break;
491 case EL_RCD:
492 if (conducting(d)) {
493 d->element[0] = el_circuit_add_resistance(c, t + RCD_L_IN, t + RCD_L_OUT, d->contact_ohm);
494 d->element[1] = el_circuit_add_resistance(c, t + RCD_N_IN, t + RCD_N_OUT, d->contact_ohm);
495 if (d->test_pressed && d->residual_rated_a > 0.0) {
496 /* The test circuit bypasses the N pole, so its current is residual current. */
497 const double ohm = 230.0 / (RCD_TEST_MULTIPLE * d->residual_rated_a);
498 el_circuit_add_resistance(c, t + RCD_L_OUT, t + RCD_N_IN, ohm);
499 }
500 }
501 break;
502 case EL_LOAD:
503 if (d->plugged_into >= 0) {
504 const int s = model->devices[d->plugged_into].first_node;
505 for (int k = 0; k < 3; k++) {
506 el_circuit_add_resistance(c, t + k, s + k, PLUG_CONTACT_OHM);
507 }
508 }
509 if (d->on) {
510 d->element[0] = el_circuit_add_resistance(c, t + LOAD_L, t + LOAD_N, load_ohm(d));
511 }
512 break;
513 case EL_DIMMER:
514 if (conducting(d) && d->level > DIMMER_OFF_LEVEL) {
515 const double x = d->level >= DIMMER_FULL_LEVEL ? 0.0 : d->reactance_ohm;
516 d->element[0] = el_circuit_add_impedance(c, t + TWO_IN, t + TWO_OUT, el_phasor(d->contact_ohm, x));
517 }
518 break;
519 case EL_SOCKET:
520 case EL_JUNCTION:
521 case EL_DEVICE_TYPES:
522 break;
523 }
524 }
525
526 if (inject_a >= 0 && inject_b >= 0) {
527 el_circuit_add_current_source(c, inject_b, inject_a, el_phasor(1.0, 0.0));
528 }
529 return el_circuit_solve(c, EL_GMIN);
530}

532static void read_results(ElModel* model) {
533 const ElCircuit* c = &model->circuit;
534 for (size_t i = 0; i < model->connection_count; i++) {
535 ElConnection* k = &model->connections[i];
536 k->current_a = k->element >= 0 ? el_abs(el_circuit_current(c, k->element)) : 0.0;
537 }
538 for (size_t i = 0; i < model->fault_count; i++) {
539 model->faults[i].current_a = el_abs(el_circuit_current(c, model->faults[i].element));
540 }
541 for (size_t i = 0; i < model->device_count; i++) {
542 ElDevice* d = &model->devices[i];
543 d->current_a = d->element[0] >= 0 ? el_abs(el_circuit_current(c, d->element[0])) : 0.0;
544 d->residual_a = 0.0;
545 d->power_w = 0.0;
546 d->voltage_v_measured = 0.0;
547 if (d->type == EL_RCD && d->element[0] >= 0) {
548 d->residual_a = el_abs(el_add(el_circuit_current(c, d->element[0]), el_circuit_current(c, d->element[1])));
549 } else if (d->type == EL_LOAD || d->type == EL_SOCKET || d->type == EL_JUNCTION) {
550 const ElPhasor u = el_sub(el_circuit_voltage(c, d->first_node + LOAD_L), el_circuit_voltage(c, d->first_node + LOAD_N));
551 d->voltage_v_measured = el_abs(u);
552 if (d->type == EL_LOAD && d->element[0] >= 0) {
553 const ElPhasor current = el_circuit_current(c, d->element[0]);
554 d->power_w = el_real(el_mul(u, el_conj(current)));
555 }
556 }
557 }

560/* Thermal state: once per step with the currents before any trip of this step. */
561static void update_heat(ElModel* model, double dt) {
562 for (size_t i = 0; i < model->device_count; i++) {
563 ElDevice* d = &model->devices[i];
564 if (d->rated_current_a <= 0.0) {
565 continue;
566 }
567 if (d->type == EL_BREAKER) {
568 const double ratio = d->current_a / d->rated_current_a;
569 d->heat = ratio > 1.0 ? d->heat + (ratio * ratio - 1.0) * dt : fmax(0.0, d->heat - BREAKER_COOLING * dt);
570 } else if (d->type == EL_FUSE) {
571 const double excess = d->current_a * d->current_a - d->rated_current_a * d->rated_current_a;
572 d->heat = excess > 0.0 ? d->heat + excess * dt
573 : fmax(0.0, d->heat - 0.1 * d->rated_current_a * d->rated_current_a * dt);
574 }
575 }
576}

578/* Checks all protective devices; returns true if one tripped. */
579static bool check_protection(ElModel* model) {
580 bool any = false;
581 for (size_t i = 0; i < model->device_count; i++) {
582 ElDevice* d = &model->devices[i];
583 if (!conducting(d)) {
584 continue;
585 }
586 if (d->type == EL_BREAKER) {
587 const double ratio = d->rated_current_a > 0.0 ? d->current_a / d->rated_current_a : 0.0;
588 if (ratio >= d->magnetic_multiple) {
589 d->tripped = true;
590 push_event(model, "%s tripped (short circuit, %.0f A = %.1f x In)", d->id, d->current_a, ratio);
591 } else if (d->heat >= d->thermal_limit) {
592 d->tripped = true;
593 push_event(model, "%s tripped (overload, %.1f A = %.2f x In)", d->id, d->current_a, ratio);
594 }
595 } else if (d->type == EL_FUSE) {
596 if (d->heat >= d->thermal_limit) {
597 d->tripped = true;
598 push_event(model, "%s blown (%.1f A)", d->id, d->current_a);
599 }
600 } else if (d->type == EL_RCD) {
601 if (d->residual_a >= RCD_TRIP_FRACTION * d->residual_rated_a) {
602 d->tripped = true;
603 push_event(model, "%s tripped (residual current %.1f mA%s)", d->id, d->residual_a * 1000.0,
604 d->test_pressed ? ", test button" : "");
605 }
606 }
607 if (d->tripped) {
608 model->revision++;
609 any = true;
610 }
611 }
612 return any;
613}
614
615/*
616 * Dimmer: an ideal (lossless) series reactance X chosen so that the current
617 * is sqrt(level) and the power of a resistive load level x the full value -
618 * the RMS values of a phase-cut or PWM dimmer. With the Thevenin impedance
619 * Z = A + jB the dimmer sees (contact included), |Z + jX| = |Z| / sqrt(level).
620 * Recomputed only when the network changed.
621 */
622static void update_dimmers(ElModel* model) {
623 for (size_t i = 0; i < model->device_count; i++) {
624 ElDevice* d = &model->devices[i];
625 if (d->type != EL_DIMMER || !conducting(d) || d->level <= DIMMER_OFF_LEVEL || d->level >= DIMMER_FULL_LEVEL) {
626 continue;
627 }
628 const int in = d->first_node + TWO_IN;
629 const int out = d->first_node + TWO_OUT;
630 if (!build_and_solve(model, true, in, out, (int)i)) {
631 continue;
632 }
633 const ElPhasor z = el_sub(el_circuit_voltage(&model->circuit, in), el_circuit_voltage(&model->circuit, out));
634 const double a = z.re + d->contact_ohm;
635 const double b = z.im;
636 const double x = -b + sqrt(fmax(0.0, (a * a + b * b) / d->level - a * a));
637 d->reactance_ohm = fmin(fmax(x, 0.0), DIMMER_MAX_REACTANCE);
638 }
639 model->dimmer_revision = model->revision;
640}
641
642bool el_model_step(ElModel* model, double dt_s) {
643 bool ok = true;
644 for (int iteration = 0; iteration < MAX_TRIP_ITERATIONS; iteration++) {
645 if (model->dimmer_revision != model->revision) {
646 update_dimmers(model);
647 }
648 if (!build_and_solve(model, false, -1, -1, -1)) {
649 ok = false;
650 break;
651 }
652 read_results(model);
653 if (iteration == 0) {
654 update_heat(model, dt_s);
655 }
656 if (!check_protection(model)) {
657 break;
658 }
659 }
660 for (size_t i = 0; i < model->device_count; i++) {
661 if (model->devices[i].test_pressed) {
662 model->devices[i].test_pressed = false;
663 model->revision++;
664 }
665 }
666 model->time_s += dt_s;
667 return ok;
668}

674bool el_model_measure_voltage(ElModel* model, const char* a, const char* b, ElPhasor* voltage) {
675 const int na = el_model_terminal_node(model, a);
676 const int nb = el_model_terminal_node(model, b);
677 if (na < 0 || nb < 0 || !build_and_solve(model, false, -1, -1, -1)) {
678 return false;
679 }
680 *voltage = el_sub(el_circuit_voltage(&model->circuit, na), el_circuit_voltage(&model->circuit, nb));
681 read_results(model);
682 return true;
683}
684
685bool el_model_measure_impedance(ElModel* model, const char* a, const char* b, ElPhasor* impedance) {
686 const int na = el_model_terminal_node(model, a);
687 const int nb = el_model_terminal_node(model, b);
688 if (na < 0 || nb < 0 || na == nb) {
689 return false;
690 }
691 /* Linear network: the Thevenin impedance does not depend on the sources. */
692 const bool ok = build_and_solve(model, true, na, nb, -1);
693 if (ok) {
694 *impedance = el_sub(el_circuit_voltage(&model->circuit, na), el_circuit_voltage(&model->circuit, nb));
695 }
696 /* Restore the normal solution for later readings. */
697 build_and_solve(model, false, -1, -1, -1);
698 read_results(model);
699 return ok;
700}
701
702bool el_model_measure_insulation(ElModel* model, const char* a, const char* b, double* ohm) {
703 ElPhasor voltage;
704 if (!el_model_measure_voltage(model, a, b, &voltage) || el_abs(voltage) > EL_SAFE_VOLTAGE) {
705 return false;
706 }
707 ElPhasor impedance;
708 if (!el_model_measure_impedance(model, a, b, &impedance)) {
709 return false;
710 }
711 *ohm = el_abs(impedance);
712 return true;
713}
