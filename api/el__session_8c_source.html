9/*
10 * Queues: single producer, single consumer rings. Positions run modulo
11 * 2 x capacity, so "full" and "empty" can be told apart without a counter.
12 *
13 * Snapshots: triple buffer. The simulation thread fills "back" and exchanges
14 * it with "middle" (marked fresh); the reader exchanges "front" with "middle"
15 * when it is fresh. Neither side ever waits.
16 */
17
18#define RING (2L * EL_SESSION_QUEUE)
19#define FRESH 0x4L
20#define INDEX_MASK 0x3L

63ElSession* el_session_create(ElModel* model) {
64 ElSession* session = (ElSession*)calloc(1, sizeof(ElSession));
65 if (session == NULL) {
66 el_model_free(model);
67 return NULL;
68 }
69 session->model = *model;
70 memset(model, 0, sizeof(*model));
71 bool ok = true;
72 for (int i = 0; i < 3; i++) {
73 ok = alloc_snapshot(&session->buffers[i], session->model.device_count, session->model.connection_count) && ok;
74 }
75 if (!ok) {
76 el_session_destroy(session);
77 return NULL;
78 }
79 session->back = 0;
80 session->middle = 1;
81 session->front = 2;
82 return session;
83}

119bool el_session_send(ElSession* session, const ElCommand* command) {
120 const long tail = el_atomic_load(&session->command_tail);
121 const long head = el_atomic_load(&session->command_head);
122 if (ring_count(head, tail) >= EL_SESSION_QUEUE) {
123 return false;
124 }
125 session->commands[tail % EL_SESSION_QUEUE] = *command;
126 el_atomic_store(&session->command_tail, (tail + 1) % RING);
127 return true;
128}

130static bool next_command(ElSession* session, ElCommand* command) {
131 const long head = el_atomic_load(&session->command_head);
132 const long tail = el_atomic_load(&session->command_tail);

134 return false;
135 }
136 *command = session->commands[head % EL_SESSION_QUEUE];
137 el_atomic_store(&session->command_head, (head + 1) % RING);

145static void publish_event(ElSession* session, const ElEvent* event) {
146 const long tail = el_atomic_load(&session->event_tail);
147 const long head = el_atomic_load(&session->event_head);
148 if (ring_count(head, tail) >= EL_SESSION_QUEUE) {
149 el_atomic_store(&session->dropped_events, el_atomic_load(&session->dropped_events) + 1);
150 return;
151 }
152 session->events[tail % EL_SESSION_QUEUE] = *event;
153 el_atomic_store(&session->event_tail, (tail + 1) % RING);
154}

197static void apply(ElSession* session, const ElCommand* c) {
198 ElModel* m = &session->model;
199 bool ok = false;
200 switch (c->type) {
201 case EL_CMD_SET_ON:
202 ok = el_model_set_on(m, c->target, c->value != 0.0);
203 break;
204 case EL_CMD_RESET:
205 ok = el_model_reset(m, c->target);
206 break;
207 case EL_CMD_PRESS_TEST:
208 ok = el_model_press_test(m, c->target);
209 break;
210 case EL_CMD_SET_LEVEL:
211 ok = el_model_set_level(m, c->target, c->value);
212 break;
213 case EL_CMD_ADD_FAULT:
214 ok = el_model_add_fault(m, c->target, c->other, c->value, NULL) >= 0;
215 break;
216 case EL_CMD_CLEAR_FAULTS:
217 el_model_clear_faults(m);
218 ok = true;
219 break;
220 case EL_CMD_SET_OPEN:
221 ok = el_model_set_connection_open(m, c->target, c->value != 0.0);
222 break;
223 case EL_CMD_PLUG:
224 ok = el_model_plug(m, c->target, c->other[0] ? c->other : NULL);
225 break;
226 }
227 if (!ok) {
228 ElEvent event;
229 event.time_s = m->time_s;
230 snprintf(event.text, sizeof(event.text), "command rejected: %s %s %s", command_name(c->type), c->target, c->other);
231 publish_event(session, &event);
232 }
233}

235static void fill_snapshot(ElSession* session, ElSnapshot* s, bool solved) {
236 const ElModel* m = &session->model;
237 s->step = session->steps;
238 s->time_s = m->time_s;
239 s->solved = solved;
240 for (size_t i = 0; i < m->device_count; i++) {
241 const ElDevice* d = &m->devices[i];
242 ElDeviceState* state = &s->devices[i];
243 state->voltage_v = d->voltage_v_measured;
244 state->current_a = d->current_a;
245 state->power_w = d->power_w;
246 state->residual_a = d->residual_a;
247 state->level = d->level;
248 state->flags = (d->on ? EL_STATE_ON : 0u) | (d->tripped ? EL_STATE_TRIPPED : 0u) |
249 (el_device_powered(d) ? EL_STATE_POWERED : 0u);
250 }
251 for (size_t i = 0; i < m->connection_count; i++) {
252 s->connection_current_a[i] = m->connections[i].current_a;
253 }

256bool el_session_step(ElSession* session, double dt_s) {
257 ElCommand command;
258 while (next_command(session, &command)) {
259 apply(session, &command);
260 }
261 const bool ok = el_model_step(&session->model, dt_s);
262 session->steps++;
263 ElEvent event;
264 while (el_model_next_event(&session->model, &event)) {
265 publish_event(session, &event);
266 }
267 fill_snapshot(session, &session->buffers[session->back], ok);
268 session->back = el_atomic_exchange(&session->middle, session->back | FRESH) & INDEX_MASK;
269 return ok;
270}

272const ElSnapshot* el_session_snapshot(ElSession* session) {
